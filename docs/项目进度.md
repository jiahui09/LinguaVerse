## 四、如何复现验证

```bash
# 一键验证（10 步：结构 + 闭环回放 + GLB 资产 + 音频离线校验 + 冒烟 + 帧采样 + 对话回归，
# 报告落 artifacts/verify-report.json，含输入 SHA256 哈希绑定与 scope 自陈）
tools/verify.sh          # 全量（约 30s–2min，Ollama 热时）
tools/verify.sh --quick  # 跳过 LLM 对话回归

# 单项：结构自检（无 GPU 可跑）
./tools/Godot_v4.7.2-stable_linux.x86_64 --headless --path godot-project \
  -s res://scripts/debug/street_check.gd

# 单项：对话链路断言（触发/走开结束/ESC 两段式/暂停不软锁）
./tools/Godot_v4.7.2-stable_linux.x86_64 --headless --path godot-project \
  -s res://scripts/debug/dialog_check.gd

# 单项：闭环回放（自动起 mock 后端，射线开对话→多轮→错误路径→走开→第二个 NPC→昼夜）
./tools/Godot_v4.7.2-stable_linux.x86_64 --headless --path godot-project \
  -s res://scripts/debug/playthrough_check.gd

# 单项：帧时间采样（headless=仅逻辑耗时；真机加 LV_PERF_ENFORCE=1 才按 p95≤20/p99≤33 判失败）
LV_ARTIFACTS_DIR="$PWD/artifacts" ./tools/Godot_v4.7.2-stable_linux.x86_64 \
  --headless --path godot-project -s res://scripts/debug/perf_drive.gd

# 单项：对话回归（固定法语脚本×轮，度量延迟与输出契约；Ollama 不可达时退出码 77=SKIP）
node tools/dialog_regression.mjs --rounds 1

# 单项：GLB 资产校验（哈希绑定，资产一变旧验收失效）
python3 tools/check_glb.py

# 单项：音频离线校验（格式/体积/无缝循环/峰值/确定性；不含主观听感）
python3 tools/check_audio.py

# 单项：重新合成环境音（确定性，字节级可复现；CC0 零下载）
python3 tools/make_ambience.py

# 启动后端 + Ollama 实测真实 AI 对话
cd backend && cp .env.example .env && pnpm install && pnpm dev

# 截图（真实 GPU）
./tools/Godot_v4.7.2-stable_linux.x86_64 --path godot-project \
  -s res://scripts/debug/screenshot_capture.gd

# 软渲染截图（本机无 GPU 时走 llvmpipe；仅作位置抽查，不能据此声称视觉验收）
XDG_DATA_HOME="$PWD/.userdata" LIBGL_ALWAYS_SOFTWARE=1 \
  ./tools/Godot_v4.7.2-stable_linux.x86_64 --rendering-driver opengl3 \
  --path godot-project -s res://scripts/debug/screenshot_capture.gd

# 直接游玩
./tools/Godot_v4.7.2-stable_linux.x86_64 --path godot-project
```
